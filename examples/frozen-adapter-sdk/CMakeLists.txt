cmake_minimum_required(VERSION 3.20)
project(RFModelFrozenAdapterExample LANGUAGES CXX)
find_package(RFModel 0.1.0 EXACT CONFIG REQUIRED)
add_executable(frozen_adapter main.cpp frozen_model.cpp)
target_link_libraries(frozen_adapter PRIVATE RFModel::rfmodel)
set_target_properties(frozen_adapter PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(frozen_adapter PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(frozen_adapter PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME frozen_adapter_workflows COMMAND frozen_adapter)
