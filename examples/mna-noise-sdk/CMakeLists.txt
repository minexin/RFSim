cmake_minimum_required(VERSION 3.20)
project(RFModelMnaNoiseExample LANGUAGES CXX)
find_package(RFModel 0.1.0 EXACT CONFIG REQUIRED)
add_executable(mna_noise main.cpp mna_noise.cpp)
target_link_libraries(mna_noise PRIVATE RFModel::rfmodel)
set_target_properties(mna_noise PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(mna_noise PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(mna_noise PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME mna_noise_isothermal COMMAND mna_noise)
add_test(NAME mna_noise_changed_rl_temperature COMMAND mna_noise --variant)
add_test(NAME mna_noise_bad_argument COMMAND mna_noise --unknown)
set_tests_properties(mna_noise_bad_argument PROPERTIES WILL_FAIL TRUE)
