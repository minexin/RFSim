cmake_minimum_required(VERSION 3.20)
project(RFModelPllPhaseJitterExample LANGUAGES CXX)
find_package(RFModel 0.1.0 EXACT CONFIG REQUIRED)
add_executable(pll_phase_jitter main.cpp pll_noise.cpp)
target_link_libraries(pll_phase_jitter PRIVATE RFModel::rfmodel)
set_target_properties(pll_phase_jitter PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(pll_phase_jitter PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(pll_phase_jitter PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME pll_phase_jitter_different_loops COMMAND pll_phase_jitter)
add_test(NAME pll_phase_jitter_equal_loops COMMAND pll_phase_jitter --equal-loops)
add_test(NAME pll_phase_jitter_bad_argument COMMAND pll_phase_jitter --unknown)
set_tests_properties(pll_phase_jitter_bad_argument PROPERTIES WILL_FAIL TRUE)
