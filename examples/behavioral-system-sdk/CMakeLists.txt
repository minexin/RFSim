cmake_minimum_required(VERSION 3.20)
project(RFModelBehavioralSystemExample LANGUAGES CXX)
find_package(RFModel 0.1.0 EXACT CONFIG REQUIRED)
add_executable(behavioral_system main.cpp workflow.cpp)
target_link_libraries(behavioral_system PRIVATE RFModel::rfmodel)
set_target_properties(behavioral_system PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(behavioral_system PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(behavioral_system PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME behavioral_system_workflows COMMAND behavioral_system)
