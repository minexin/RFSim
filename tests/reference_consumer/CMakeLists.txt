cmake_minimum_required(VERSION 3.20)
project(RFModelReferenceConsumer LANGUAGES CXX)
find_package(RFModel CONFIG REQUIRED)
# Both translation units are copied into this project by the SDK check. The
# public installed target is the sole source of RFModel include directories.
add_executable(installed_reference_probe reference_highorder_probe.cpp)
target_link_libraries(installed_reference_probe PRIVATE RFModel::rfmodel)
add_executable(installed_reference_oracle reference_highorder_oracle.cpp)
target_link_libraries(installed_reference_oracle PRIVATE RFModel::rfmodel)
set_target_properties(installed_reference_probe installed_reference_oracle
    PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
get_target_property(sdk_includes RFModel::rfmodel INTERFACE_INCLUDE_DIRECTORIES)
file(GLOB_RECURSE headers "${sdk_includes}/rfmodel/*.hpp")
list(APPEND headers "${CMAKE_CURRENT_SOURCE_DIR}/reference_highorder_probe.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/reference_highorder_oracle.cpp")
list(SORT headers)
set(files "")
set(content "")
foreach(path IN LISTS headers)
    file(SHA256 "${path}" hash)
    if(path MATCHES "/rfmodel/(.+)$")
        set(key "installed/rfmodel/${CMAKE_MATCH_1}")
    else()
        get_filename_component(key "${path}" NAME)
    endif()
    string(APPEND content "${key} ${hash}\n")
    if(files)
        string(APPEND files ",\n")
    endif()
    string(APPEND files "    \"${key}\": \"${hash}\"")
endforeach()
string(SHA256 content_hash "${content}")
if(CMAKE_CONFIGURATION_TYPES)
    set(configs ${CMAKE_CONFIGURATION_TYPES})
else()
    set(configs "${CMAKE_BUILD_TYPE}")
endif()
foreach(config IN LISTS configs)
    string(TOUPPER "${config}" uppercase)
    set(flags "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${uppercase}}; C++17; extensions=OFF")
    string(REPLACE "\\" "\\\\" flags "${flags}")
    string(REPLACE "\"" "\\\"" flags "${flags}")
    set(identity "  \"protocol\": 1,\n  \"source_revision\": \"${SOURCE_REVISION}\",\n  \"source_content_sha256\": \"${content_hash}\",\n  \"configuration\": \"${config}\",\n  \"compiler\": {\"id\": \"${CMAKE_CXX_COMPILER_ID}\", \"version\": \"${CMAKE_CXX_COMPILER_VERSION}\", \"flags\": \"${flags}\"},\n  \"installed_sdk_consumer\": true,\n  \"source_files\": {\n${files}\n  }")
    string(SHA256 build_id "${identity}")
    target_compile_definitions(installed_reference_probe PRIVATE
        "$<$<CONFIG:${config}>:RFMODEL_NATIVE_REPLAY_BUILD_ID=\"${build_id}\">")
    file(GENERATE OUTPUT "$<TARGET_FILE:installed_reference_probe>.manifest.json"
        CONTENT "{\n  \"build_id\": \"${build_id}\",\n${identity}\n}\n"
        CONDITION "$<CONFIG:${config}>" NEWLINE_STYLE LF)
endforeach()
enable_testing()
add_test(NAME installed_probe_protocol COMMAND "${PYTHON}" -I -S "${PROTOCOL_TEST}"
    $<TARGET_FILE:installed_reference_probe> $<TARGET_FILE:installed_reference_oracle>)
