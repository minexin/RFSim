cmake_minimum_required(VERSION 3.20)
project(RFModelConsumer LANGUAGES CXX)
find_package(RFModel 0.1.0 EXACT CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE RFModel::rfmodel)
enable_testing()
add_test(NAME installed_consumer COMMAND consumer)
if(TARGET RFModel::c_api)
    enable_language(C)
    add_executable(c_consumer c_main.c)
    target_link_libraries(c_consumer PRIVATE RFModel::c_api)
    if(UNIX)
        target_link_libraries(c_consumer PRIVATE m)
    endif()
    if(WIN32)
        add_custom_command(TARGET c_consumer POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:RFModel::c_api> $<TARGET_FILE_DIR:c_consumer>)
    endif()
    add_test(NAME installed_c_consumer COMMAND c_consumer "${CMAKE_CURRENT_SOURCE_DIR}/matched.s1p")
endif()
