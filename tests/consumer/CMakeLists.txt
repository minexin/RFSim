cmake_minimum_required(VERSION 3.20)
project(RFModelConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(RFModel 0.1.0 EXACT CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE RFModel::rfmodel)
enable_testing()
add_executable(model_adapter_consumer model_adapter.cpp model_adapter_probe.cpp)
target_link_libraries(model_adapter_consumer PRIVATE RFModel::rfmodel)
add_test(NAME installed_model_adapter_consumer COMMAND model_adapter_consumer)
add_test(NAME installed_consumer COMMAND consumer)
add_executable(conversion_api_consumer conversion_api.cpp conversion_api_probe.cpp)
target_link_libraries(conversion_api_consumer PRIVATE RFModel::rfmodel)
add_test(NAME installed_conversion_api_consumer COMMAND conversion_api_consumer)
add_executable(operating_power_wave_consumer operating_power_wave.cpp operating_power_wave_probe.cpp)
target_link_libraries(operating_power_wave_consumer PRIVATE RFModel::rfmodel)
add_test(NAME installed_operating_power_wave_consumer COMMAND operating_power_wave_consumer)
add_executable(coherent_pipeline_consumer coherent_pipeline.cpp coherent_pipeline_probe.cpp)
target_link_libraries(coherent_pipeline_consumer PRIVATE RFModel::rfmodel)
add_test(NAME installed_coherent_pipeline_consumer COMMAND coherent_pipeline_consumer)
if(TARGET RFModel::c_api)
    enable_language(C)
    add_executable(c_consumer c_main.c)
    target_link_libraries(c_consumer PRIVATE RFModel::c_api)
    if(UNIX)
        target_link_libraries(c_consumer PRIVATE m)
    endif()
    if(WIN32)
        add_custom_command(TARGET c_consumer POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:RFModel::c_api> $<TARGET_FILE_DIR:c_consumer>)
    endif()
    add_test(NAME installed_c_consumer COMMAND c_consumer "${CMAKE_CURRENT_SOURCE_DIR}/matched.s1p")
endif()
