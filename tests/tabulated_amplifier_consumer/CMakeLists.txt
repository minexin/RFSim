cmake_minimum_required(VERSION 3.20)
project(TabulatedAmplifierConsumer LANGUAGES CXX)
find_package(RFModel CONFIG REQUIRED)
add_executable(consumer main.cpp second.cpp)
target_link_libraries(consumer PRIVATE RFModel::rfmodel)
set_target_properties(consumer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(consumer PRIVATE -pedantic-errors)
endif()
enable_testing()
add_test(NAME consumer COMMAND consumer)
