cmake_minimum_required(VERSION 3.20)
project(InstalledXParameterReplay LANGUAGES CXX)
if(NOT IS_ABSOLUTE "${SDK_PREFIX}")
    message(FATAL_ERROR "Explicit absolute SDK_PREFIX required")
endif()
file(REAL_PATH "${SDK_PREFIX}" sdk_real)
set(config_dir "${sdk_real}/lib/cmake/RFModel")
find_package(RFModel 0.1.0 EXACT CONFIG REQUIRED PATHS "${config_dir}" NO_DEFAULT_PATH)
file(REAL_PATH "${RFModel_DIR}" found_config)
if(NOT found_config STREQUAL config_dir)
    message(FATAL_ERROR "Unexpected RFModel configuration")
endif()
get_target_property(includes RFModel::rfmodel INTERFACE_INCLUDE_DIRECTORIES)
foreach(include_dir IN LISTS includes)
    file(REAL_PATH "${include_dir}" include_real)
    if(NOT include_real STREQUAL "${sdk_real}/include")
        message(FATAL_ERROR "Unexpected SDK include; source fallback prohibited")
    endif()
endforeach()
if(NOT includes)
    message(FATAL_ERROR "Installed SDK has no includes")
endif()
add_executable(xparameter_replay_probe probe-xparameter-data.cpp odr.cpp)
target_link_libraries(xparameter_replay_probe PRIVATE RFModel::rfmodel)
if(CMAKE_CXX_COMPILER_ARG1)
    message(FATAL_ERROR "Compiler argument injection is prohibited")
endif()
file(REAL_PATH "${sdk_real}/include/rfmodel/io/xparameter_data.hpp" reader_header)
file(RELATIVE_PATH reader_relative "${sdk_real}" "${reader_header}")
if(reader_relative MATCHES "^\\.\\.")
    message(FATAL_ERROR "Reader header escapes installed SDK")
endif()
# Absolute header selection binds the actual compilation to the recorded file.
target_compile_definitions(xparameter_replay_probe PRIVATE
    RFMODEL_XPARAMETER_INSTALLED_HEADER="${reader_header}")
target_compile_features(xparameter_replay_probe PRIVATE cxx_std_17)
set_target_properties(xparameter_replay_probe PROPERTIES CXX_EXTENSIONS OFF
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(xparameter_replay_probe PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/bin")
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/compiler-path.txt" "${CMAKE_CXX_COMPILER}")
file(WRITE "${CMAKE_BINARY_DIR}/compiler-id.txt" "${CMAKE_CXX_COMPILER_ID}")
file(WRITE "${CMAKE_BINARY_DIR}/compiler-version.txt" "${CMAKE_CXX_COMPILER_VERSION}")
file(WRITE "${CMAKE_BINARY_DIR}/reader-header.txt" "${reader_header}")
