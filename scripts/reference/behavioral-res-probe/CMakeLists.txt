cmake_minimum_required(VERSION 3.20)
project(RFModelBehavioralResProbe LANGUAGES CXX)

# Developer evidence consumer: every RFModel include must resolve inside a
# deliberately installed SDK. Package registry/source-tree fallbacks are errors.
if(NOT IS_ABSOLUTE "${SDK_PREFIX}")
    message(FATAL_ERROR "Explicit absolute SDK_PREFIX required")
endif()
file(REAL_PATH "${SDK_PREFIX}" sdk_real)
set(config_dir "${sdk_real}/lib/cmake/RFModel")
find_package(RFModel 0.1.0 EXACT CONFIG REQUIRED PATHS "${config_dir}" NO_DEFAULT_PATH)
file(REAL_PATH "${RFModel_DIR}" found_config)
if(NOT found_config STREQUAL config_dir)
    message(FATAL_ERROR "Unexpected RFModel configuration")
endif()
get_target_property(includes RFModel::rfmodel INTERFACE_INCLUDE_DIRECTORIES)
if(NOT includes)
    message(FATAL_ERROR "Installed SDK has no includes")
endif()
foreach(include_dir IN LISTS includes)
    file(REAL_PATH "${include_dir}" include_real)
    if(NOT include_real STREQUAL "${sdk_real}/include")
        message(FATAL_ERROR "Unexpected SDK include; source fallback prohibited")
    endif()
endforeach()
if(CMAKE_CXX_COMPILER_ARG1)
    message(FATAL_ERROR "Compiler argument injection is prohibited")
endif()
set(headers adapters/builtin_model_adapters.hpp conversion/conversion_simulation.hpp
    linear/linear_analysis.hpp linear/linear_path.hpp)
set(installed_header_contents "// Generated installed-only includes.\n#pragma once\n")
foreach(header IN LISTS headers)
    file(REAL_PATH "${sdk_real}/include/rfmodel/${header}" header_real)
    file(RELATIVE_PATH header_relative "${sdk_real}" "${header_real}")
    if(header_relative MATCHES "^\\.\\." OR NOT EXISTS "${header_real}")
        message(FATAL_ERROR "Required header absent or outside installed SDK: ${header}")
    endif()
    string(APPEND installed_header_contents "#include \"${header_real}\"\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/installed_headers.hpp" "${installed_header_contents}")
add_executable(behavioral_res_probe main.cpp fixture.cpp)
target_link_libraries(behavioral_res_probe PRIVATE RFModel::rfmodel)
target_include_directories(behavioral_res_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(behavioral_res_probe PROPERTIES
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set_target_properties(behavioral_res_probe PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/bin")
endforeach()
if(MSVC)
    target_compile_options(behavioral_res_probe PRIVATE /W4 /WX /permissive- /showIncludes)
else()
    target_compile_options(behavioral_res_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
file(WRITE "${CMAKE_BINARY_DIR}/compiler-path.txt" "${CMAKE_CXX_COMPILER}")
file(WRITE "${CMAKE_BINARY_DIR}/compiler-id.txt" "${CMAKE_CXX_COMPILER_ID}")
file(WRITE "${CMAKE_BINARY_DIR}/compiler-version.txt" "${CMAKE_CXX_COMPILER_VERSION}")
file(WRITE "${CMAKE_BINARY_DIR}/sdk-prefix.txt" "${sdk_real}")
